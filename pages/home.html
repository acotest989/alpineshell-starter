Every command there is

my-app — bash 
$ npx degit acotest989/alpineshell-starter my-app             # or "Use this template" on GitHub
$ cd my-app/server

$ ./setup.sh                                                  # downloads PocketBase — .\setup.ps1 on Windows
$ ./pocketbase serve --publicDir=..                           # app and API on http://127.0.0.1:8090
$ ./pocketbase superuser create you@example.com yourpassword  # first run only, for the /_/ dashboard

$ docker build -f server/Dockerfile --build-arg PB_VERSION=$(cat server/.pb-version) -t my-app .
$ docker run -p 8090:8090 -v pb_data:/pb/pb_data my-app       # the volume is the deployment
